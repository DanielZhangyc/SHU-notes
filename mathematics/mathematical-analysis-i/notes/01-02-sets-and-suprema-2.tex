\section{确界原理的证明}
\begin{theorem}[确界原理]
非空、有上界的实数集必有上确界；非空、有下界的实数集必有下确界。
\end{theorem}

下面先给出上确界存在性的两种证明，分别基于实数的十进制无限小数表示与闭区间套定理；随后通过取相反数集合，将下确界的存在性直接归结为上确界的情形。

\subsection{十进制构造}
\begin{proof}
设非空实数集 $S$ 有上界，以实数的无限小数表示为基础逐位构造其上确界。

\textbf{第一步：平移归约。}任取 $s_0\in S$，将 $S$ 中每个元素都加上 $1-s_0$。平移后的集合依然有上界，且至少包含正数 $1$；若它有上确界，将该上确界减去 $1-s_0$ 即得原集合的上确界。因此不妨设 $S$ 含有正数。

\medskip
\textbf{第二步：逐位构造小数。}因为 $S$ 含有正数，$0$ 不是 $S$ 的上界；又因 $S$ 有上界，充分大的整数都是 $S$ 的上界。于是，不是 $S$ 上界的非负整数构成一个非空有限集，从而存在最大元 $a_0$。记 $p_0=a_0$，则 $p_0$ 不是上界，而 $p_0+1$ 是上界。

一般地，设 $p_{k-1}$ 已确定且满足 $p_{k-1}$ 不是上界、$p_{k-1}+10^{-(k-1)}$ 是上界。将区间 $[p_{k-1},p_{k-1}+10^{-(k-1)}]$ 十等分，在数字 $0,1,\ldots,9$ 中选取最大者 $a_k$，使
\[
  p_k=p_{k-1}+a_k 10^{-k}
\]
不是 $S$ 的上界。由于 $p_{k-1}$ 本身不是上界，数字 $0$ 至少满足要求，故 $a_k$ 必存在。再由 $a_k$ 的最大性（当 $a_k=9$ 时，结合上一层 $p_{k-1}+10^{-(k-1)}$ 为上界的结论），对每个 $k\ge0$ 均有
\[
  \exists x\in S:\ x>p_k,
  \qquad \forall x\in S:\ x\le p_k+10^{-k}.
\]
由此定义实数 $\eta=a_0.a_1a_2\cdots$，则对所有 $k\ge0$ 都有 $p_k\le\eta\le p_k+10^{-k}$。考虑到上述构造可能出现末尾循环 $9$ 的情形，右端保留非严格不等号 $\le$。

\medskip
\textbf{第三步：验证 $\eta=\sup S$。}先证 $\eta$ 是 $S$ 的上界。若存在 $x\in S$ 满足 $x>\eta$，则可取充分大的 $k$ 使 $10^{-k}<x-\eta$，于是
\[
  x\le p_k+10^{-k}\le\eta+10^{-k}<x,
\]
产生矛盾，故 $\eta$ 必为上界。再证 $\eta$ 是最小上界：任取 $\alpha<\eta$，取充分大的 $k$ 使 $10^{-k}<\eta-\alpha$，便有
\[
  p_k\ge\eta-10^{-k}>\alpha.
\]
由于 $p_k$ 不是上界，存在 $x\in S$ 使 $x>p_k>\alpha$。这说明任何小于 $\eta$ 的实数都不是上界，因此 $\eta=\sup S$。
\end{proof}

\subsection{闭区间套证明}
\noindent 若以闭区间套定理为前提（即逐层包含且长度趋于 $0$ 的非空闭区间列 $[a_k,b_k]$ 恰有一个公共点），则可通过不断二分区间直接构造上确界。

\begin{proof}
设 $S\ne\varnothing$ 且有上界。任取 $x_0\in S$，令 $a_0=x_0-1$，再取 $S$ 的一个上界 $b_0$，则 $a_0$ 不是上界，而 $b_0$ 是上界。

\textbf{第一步：二分构造闭区间套。}设 $[a_k,b_k]$ 已确定，取其中点 $m_k=(a_k+b_k)/2$，并按 $m_k$ 是否为上界定义下一层区间：
\[
  [a_{k+1},b_{k+1}]=
  \begin{cases}
    [a_k,m_k],&m_k\text{ 是 }S\text{ 的上界},\\
    [m_k,b_k],&m_k\text{ 不是 }S\text{ 的上界}.
  \end{cases}
\]
由归纳法可知，对所有 $k\ge0$ 都有
\[
  a_k\text{ 不是上界},\qquad b_k\text{ 是上界},\qquad
  b_k-a_k=\frac{b_0-a_0}{2^k}.
\]
这些闭区间逐层包含，且长度随 $k$ 增大趋于 $0$。由闭区间套定理，存在唯一实数 $\xi$，使得对所有 $k\ge0$ 都有 $a_k\le\xi\le b_k$。

\medskip
\textbf{第二步：验证 $\xi=\sup S$。}若存在 $x\in S$ 满足 $x>\xi$，取充分大的 $k$ 使 $b_k-a_k<x-\xi$，则
\[
  b_k-\xi\le b_k-a_k<x-\xi,
\]
从而 $b_k<x$，与 $b_k$ 是上界矛盾，故 $\xi$ 是上界。

再任取 $\alpha<\xi$，取充分大的 $k$ 使 $b_k-a_k<\xi-\alpha$，则
\[
  \xi-a_k\le b_k-a_k<\xi-\alpha,
  \qquad \text{即 }\alpha<a_k.
\]
由于 $a_k$ 不是上界，存在 $x\in S$ 使 $x>a_k>\alpha$。因此任何小于 $\xi$ 的实数都不是上界，即 $\xi=\sup S$。
\end{proof}

\medskip
\noindent\textbf{下确界的存在性。}
若非空数集 $S$ 有下界，则其相反数集合 $-S=\{-x:x\in S\}$ 非空且有上界。由已证结论，$\beta=\sup(-S)$ 存在。对任意 $x\in S$，由 $-x\le\beta$ 得 $x\ge-\beta$，故 $-\beta$ 是 $S$ 的下界；又对任意 $\varepsilon>0$，存在 $-x\in-S$ 使 $\beta-\varepsilon<-x$，即 $x<-\beta+\varepsilon$，满足下确界的任意逼近性。因此 $\inf S=-\beta$，确界原理的另一半也得证。

\section{确界的性质}
确界存在性建立之后，下面进一步讨论确界与最值的关系、数集之间的序比较，以及平移和正数倍变换对确界的影响。

\begin{proposition}[最大值与上确界]
设 $S\subseteq\R$ 非空，则
\[
  \max S=\eta
  \quad\Longleftrightarrow\quad
  \eta=\sup S\ \text{且}\ \eta\in S.
\]
\end{proposition}
\begin{proof}
若 $\max S=\eta$，则 $\eta\in S$，且对所有 $x\in S$ 都有 $x\le\eta$，说明 $\eta$ 是 $S$ 的上界。进一步，对任意 $\alpha<\eta$，直接取 $x=\eta\in S$ 即有 $x>\alpha$，故 $\eta$ 是最小上界，即 $\sup S=\eta$。

反之，若 $\eta=\sup S$ 且 $\eta\in S$，则由上确界的上界性可知，对所有 $x\in S$ 都有 $x\le\eta$，因此 $\max S=\eta$。
\end{proof}

\noindent 由此可见，仅有 $\sup S=\eta$ 并不足以推出 $S$ 存在最大值，还必须要求 $\eta\in S$。

\begin{proposition}[两个数集的确界比较]
设 $A,B\subseteq\R$ 均非空，且
\[
  \forall x\in A,\quad\forall y\in B,\quad x\le y.
\]
则 $A$ 有上确界，$B$ 有下确界，并且
\[
  \sup A\le\inf B.
\]
\end{proposition}
\begin{proof}
任取 $y_0\in B$，由题设可知 $y_0$ 是 $A$ 的上界；同理，任取 $x_0\in A$，则 $x_0$ 是 $B$ 的下界。根据确界原理，$\sup A$ 与 $\inf B$ 均存在。

进一步，对任意 $y\in B$，$y$ 都是 $A$ 的上界。因为 $\sup A$ 是 $A$ 的最小上界，故有
\[
  \sup A\le y\qquad(\forall y\in B).
\]
这表明 $\sup A$ 是 $B$ 的一个下界。再由 $\inf B$ 是 $B$ 的最大下界，即得 $\sup A\le\inf B$。
\end{proof}

\begin{proposition}[平移与正数倍]
设 $S\subseteq\R$ 非空且有上界，记 $\eta=\sup S$。
\begin{enumerate}
  \item 对任意 $a\in\R$，定义 $S+a=\{x+a:x\in S\}$，则
  \[
    \sup(S+a)=\sup S+a.
  \]
  \item 对任意 $b>0$，定义 $bS=\{bx:x\in S\}$，则
  \[
    \sup(bS)=b\sup S.
  \]
\end{enumerate}
\end{proposition}
\begin{proof}
\textbf{（1）平移。}
对任意 $x\in S$，由 $x\le\eta$ 可得 $x+a\le\eta+a$，因此 $\eta+a$ 是 $S+a$ 的上界。

为验证最小性，任取 $\alpha<\eta+a$，则 $\alpha-a<\eta$。由 $\eta=\sup S$ 可知，存在 $x_0\in S$ 使得
\[
  x_0>\alpha-a,
  \qquad \text{从而 }x_0+a>\alpha.
\]
因为 $x_0+a\in S+a$，任何小于 $\eta+a$ 的实数都不是 $S+a$ 的上界，故 $\sup(S+a)=\eta+a$。

\medskip
\textbf{（2）正数倍。}
由于 $b>0$，对任意 $x\in S$，由 $x\le\eta$ 两边同乘 $b$ 得 $bx\le b\eta$，因此 $b\eta$ 是 $bS$ 的上界。

再任取 $\alpha<b\eta$，则 $\alpha/b<\eta$。由 $\eta=\sup S$ 可知，存在 $x_0\in S$ 使得
\[
  x_0>\frac\alpha b,
  \qquad \text{从而 }bx_0>\alpha.
\]
因为 $bx_0\in bS$，任何小于 $b\eta$ 的实数都不是 $bS$ 的上界，故 $\sup(bS)=b\eta$。
\end{proof}

\subsection{习题 1.2，第 5 题}
\noindent\textbf{题目：}设 $S$ 为非空有下界数集，证明
\[
  \inf S=\xi\in S\quad\Longleftrightarrow\quad\xi=\min S.
\]

\begin{proof}
先证必要性：设 $\xi=\inf S\in S$。由下确界的下界性，对任意 $x\in S$ 都有 $x\ge\xi$；又因 $\xi\in S$，故 $\min S=\xi=\inf S$。

再证充分性：若 $S$ 有最小值，记 $\xi=\min S$，则 $\xi\in S$ 且对所有 $x\in S$ 都有 $x\ge\xi$，因此 $\xi$ 是 $S$ 的下界。对任意 $\beta>\xi$，直接取 $x=\xi\in S$ 便有 $x<\beta$，说明任何大于 $\xi$ 的实数都不是 $S$ 的下界。于是 $\inf S=\xi\in S$。

由此两向均得证，且最小值存在时必等于下确界。
\end{proof}

\subsection{习题 1.2，第 6 题（1）}
\noindent\textbf{题目：}设 $S$ 为非空数集，定义
$S^{-}=\{x:-x\in S\}$。证明
\[
  \inf S^{-}=-\sup S.
\]
\noindent 这里按有限实数确界讨论，即假设 $S$ 有上界以保证 $\sup S$ 存在。

\begin{proof}
记 $M=\sup S$。任取 $y\in S^{-}$，按定义有 $-y\in S$，故
\[
  -y\le M\quad\Longrightarrow\quad y\ge-M,
\]
这表明 $-M$ 是 $S^{-}$ 的一个下界。

为验证 $-M$ 是最大下界，任取 $\beta>-M$，则 $-\beta<M$。由 $M=\sup S$，存在 $x_0\in S$ 使得
\[
  x_0>-\beta\quad\Longrightarrow\quad -x_0<\beta.
\]
由于 $-x_0\in S^{-}$，任何大于 $-M$ 的实数都不是 $S^{-}$ 的下界，从而
\[
  \inf S^{-}=-M=-\sup S.
\]
\end{proof}

\subsection{习题 1.2，第 7 题（1）}
\noindent\textbf{题目：}设 $A,B$ 皆为非空有界数集，定义
\[
  A+B=\{x+y:x\in A,\ y\in B\}.
\]
证明
\[
  \sup(A+B)=\sup A+\sup B.
\]

\begin{proof}
记 $M_A=\sup A$，$M_B=\sup B$。由 $A,B$ 均非空可知，$A+B$ 也非空。

\textbf{第一步：验证上界性。}任取 $z\in A+B$，则存在 $x\in A$ 与 $y\in B$ 使 $z=x+y$。由上确界的上界性得 $x\le M_A$、$y\le M_B$，相加即有
\[
  z=x+y\le M_A+M_B,
\]
故 $M_A+M_B$ 是 $A+B$ 的一个上界。

\medskip
\textbf{第二步：验证它是最小上界。}任取 $c<M_A+M_B$，移项得 $c-M_B<M_A$。由 $M_A=\sup A$，存在 $x_0\in A$ 使得
\[
  x_0>c-M_B,
\]
再次移项即得 $c-x_0<M_B$。于是再由 $M_B=\sup B$，存在 $y_0\in B$ 使得
\[
  y_0>c-x_0.
\]
由此 $x_0+y_0\in A+B$ 且 $x_0+y_0>c$，表明 $c$ 不是 $A+B$ 的上界。综上，任何小于 $M_A+M_B$ 的实数都不是 $A+B$ 的上界，从而
\[
  \sup(A+B)=M_A+M_B=\sup A+\sup B.
\]
\end{proof}
