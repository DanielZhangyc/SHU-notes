\section{收敛数列的性质}

以下均设 $\{x_n\}$ 为实数列。若存在 $a\in\R$ 使
\[
  \forall\varepsilon>0,\quad \exists N\in\Nplus,\quad
  \forall n>N,\quad |x_n-a|<\varepsilon,
\]
则记 $x_n\to a$。

\begin{theorem}[唯一性]
若 $x_n\to a$ 且 $x_n\to b$，则 $a=b$。
\end{theorem}
\begin{proof}
反设 $a\ne b$。取 $\varepsilon=|a-b|/3>0$。由 $x_n\to a$，存在
$N_1$ 使 $n>N_1$ 时 $|x_n-a|<\varepsilon$；由 $x_n\to b$，存在
$N_2$ 使 $n>N_2$ 时 $|x_n-b|<\varepsilon$。
取 $n>\max\{N_1,N_2\}$，由三角不等式得
\[
 |a-b|\le |a-x_n|+|x_n-b|<2\varepsilon
 =\frac23|a-b|,
\]
矛盾。因此 $a=b$。
\end{proof}

\begin{theorem}[有界性]
若 $x_n\to a$，则数列 $\{x_n\}$ 有界，即存在 $M>0$，使得对一切
$n\in\Nplus$ 都有 $|x_n|\le M$。
\end{theorem}
\begin{proof}
取 $\varepsilon=1$。由 $x_n\to a$，存在 $N\in\Nplus$，使 $n>N$ 时
\[
 |x_n-a|<1.
\]
于是对 $n>N$，
\[
 |x_n|\le |x_n-a|+|a|<1+|a|.
\]
前 $N$ 项只有有限项，令
\[
 M=\max\bigl\{|x_1|,\ldots,|x_N|,1+|a|\bigr\},
\]
则对所有 $n\in\Nplus$ 都有 $|x_n|\le M$，故数列有界。
\end{proof}

\newpage
\begin{theorem}[保号性]
若 $x_n\to a$ 且 $a>0$，则存在 $N\in\Nplus$，使 $n>N$ 时 $x_n>0$。
若 $a<0$，则存在 $N\in\Nplus$，使 $n>N$ 时 $x_n<0$。
\end{theorem}
\begin{proof}
先设 $a>0$。在极限定义中取 $\varepsilon=a/2$，则存在 $N$，使 $n>N$ 时
\[
 |x_n-a|<\frac a2.
\]
因此
\[
 x_n\ge a-|x_n-a|>a-\frac a2=\frac a2>0.
\]
若 $a<0$，取 $\varepsilon=|a|/2$，则
\[
 x_n\le a+|x_n-a|<a+\frac{|a|}{2}=\frac a2<0
\]
对充分大的 $n$ 成立。
\end{proof}

\subsection*{最终区间性质}
保号性可推广为：若 $x_n\to a$，且 $b<a<c$，则存在 $N\in\Nplus$，使
\[
 n>N\quad\Longrightarrow\quad b<x_n<c.
\]
证明时取
\[
 \varepsilon=\min\{a-b,c-a\}>0.
\]
当 $n>N$ 时，$|x_n-a|<\varepsilon$，于是
\[
 b\le a-\varepsilon<x_n<a+\varepsilon\le c.
\]
因此，收敛数列从某一项起始终落在任意包含其极限的开区间内。

\newpage
\subsection*{极限与大小关系}
若 $x_n\to a$、$y_n\to b$ 且 $a<b$，则充分大的 $n$ 都有 $x_n<y_n$。
事实上，取 $\varepsilon=(b-a)/3$，当 $n$ 充分大时，
\[
 x_n<a+\varepsilon<b-\varepsilon<y_n.
\]

\begin{theorem}[保序性]
设 $x_n\to a$、$y_n\to b$，且存在 $N_0$，使 $n>N_0$ 时
$x_n\le y_n$，则 $a\le b$。
\end{theorem}
\begin{proof}
若 $a>b$，由上面的结论，充分大的 $n$ 都有 $x_n>y_n$，与条件矛盾。
故 $a\le b$。
\end{proof}
特别地，若 $x_n\ge0$ 从某一项起成立，则 $a\ge0$。
严格不等式取极限后不一定仍严格。例如 $0<1/n$，但两边极限均为 $0$。

\begin{theorem}[夹逼定理]
设从某一项起 $x_n\le y_n\le z_n$，且 $x_n\to a$、$z_n\to a$，
则 $y_n\to a$。
\end{theorem}
\begin{proof}
任取 $\varepsilon>0$。存在 $N$，使 $n>N$ 时上述大小关系成立，且
$|x_n-a|<\varepsilon$、$|z_n-a|<\varepsilon$。于是
\[
 a-\varepsilon<x_n\le y_n\le z_n<a+\varepsilon,
\]
即 $|y_n-a|<\varepsilon$，故 $y_n\to a$。
\end{proof}

\newpage
\begin{theorem}[四则运算法则]
若 $x_n\to a$、$y_n\to b$，则
\[
 x_n\pm y_n\to a\pm b,\qquad x_ny_n\to ab.
\]
若另有 $b\ne0$，则 $y_n$ 从某一项起不为零，且
\[
 \frac{x_n}{y_n}\longrightarrow\frac ab.
\]
\end{theorem}
\begin{proof}
\textbf{加减法。} 由
\[
 |(x_n\pm y_n)-(a\pm b)|\le |x_n-a|+|y_n-b|,
\]
只需将右侧两项分别控制在 $\varepsilon/2$ 内即可。

\textbf{乘法。} 由收敛数列的有界性，存在 $M>0$，使所有 $n$ 都有
$|x_n|\le M$。于是
\[
 |x_ny_n-ab|\le |x_n|\,|y_n-b|+|b|\,|x_n-a|
 \le M|y_n-b|+|b|\,|x_n-a|.
\]
任取 $\varepsilon>0$，当 $n$ 充分大时，两项误差均小于
$\varepsilon/(M+|b|+1)$，从而 $|x_ny_n-ab|<\varepsilon$。

\textbf{除法。} 由 $y_n\to b\ne0$，充分大的 $n$ 满足
$|y_n-b|<|b|/2$，从而 $|y_n|>|b|/2>0$。因此
\[
 \left|\frac1{y_n}-\frac1b\right|
 =\frac{|y_n-b|}{|b|\,|y_n|}
 \le\frac2{|b|^2}|y_n-b|\longrightarrow0.
\]
再用乘法法则，即得 $x_n/y_n\to a/b$。
\end{proof}
四则运算法则可逐次用于固定有限个收敛数列；若运算的项数随 $n$ 增长，
则不能直接逐项取极限。
