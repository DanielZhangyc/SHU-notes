\section{函数极限的性质}

以下各定理均在同一种趋近方式下讨论，例如 $x\to x_0$、$x\to x_0^+$ 或 $x\to+\infty$。为简洁起见，正文先写 $x\to x_0$；单侧极限和无穷远处的结论完全相同，只需把邻域条件换成相应的单侧或远端条件。

\subsection*{唯一性}
\begin{theorem}[唯一性]
若
\[
 \lim_{x\to x_0}f(x)=A,\qquad \lim_{x\to x_0}f(x)=B,
\]
则 $A=B$。
\end{theorem}
\begin{proof}
反设 $A\ne B$，取 $\varepsilon=|A-B|/3>0$。由两个极限的定义，存在 $\delta_1,\delta_2>0$，使得当 $0<|x-x_0|<\delta_i$ 时，分别有
\[
 |f(x)-A|<\varepsilon,\qquad |f(x)-B|<\varepsilon.
\]
取 $\delta=\min\{\delta_1,\delta_2\}$，任取 $0<|x-x_0|<\delta$，则
\[
 |A-B|\le |A-f(x)|+|f(x)-B|<2\varepsilon
 =\frac23|A-B|,
\]
矛盾。故 $A=B$。
\end{proof}

\subsection*{局部有界性}
\begin{theorem}[局部有界性]
若 $\lim_{x\to x_0}f(x)=A$，则存在 $\delta>0$ 和 $M>0$，使得当
\[
 0<|x-x_0|<\delta
\]
时有 $|f(x)|<M$。
\end{theorem}
\begin{proof}
在极限定义中取 $\varepsilon=1$，可得某个 $\delta>0$，使
\[
 |f(x)-A|<1
\]
对所有 $0<|x-x_0|<\delta$ 成立。因此
\[
 |f(x)|\le |f(x)-A|+|A|<1+|A|.
\]
取 $M=1+|A|$ 即可。
\end{proof}

这里的“局部”是指某个去心邻域；极限存在并不要求函数在整个定义域上有界。局部有界性本身也不能推出极限存在，例如 $\sin(1/x)$ 在 $x\to0$ 时有界但没有极限。

\subsection*{保号性}
\begin{theorem}[保号性]
若 $\lim_{x\to x_0}f(x)=A>0$，则存在 $\delta>0$，使 $0<|x-x_0|<\delta$ 时 $f(x)>0$。若 $A<0$，则在某个去心邻域内 $f(x)<0$。
\end{theorem}
\begin{proof}
设 $A>0$。取 $\varepsilon=A/2$，则存在 $\delta>0$，使
\[
 |f(x)-A|<\frac A2.
\]
于是
\[
 f(x)>A-\frac A2=\frac A2>0
\]
对所有 $0<|x-x_0|<\delta$ 成立。

当 $A<0$ 时，取 $\varepsilon=|A|/2$，可得
\[
 f(x)<A+\frac{|A|}{2}=\frac A2<0.
\]
\end{proof}

保号性只适用于极限严格大于零或严格小于零的情形。若极限为零，函数值可以在邻域内变号，例如 $f(x)=x$ 在 $x\to0$ 时既有正值也有负值。

\subsection*{保序性与夹逼定理}
\begin{theorem}[保序性]
若 $\lim_{x\to x_0}f(x)=A$、$\lim_{x\to x_0}g(x)=B$，并且存在 $\delta_0>0$，使
\[
 f(x)\le g(x)\qquad(0<|x-x_0|<\delta_0),
\]
则 $A\le B$。
\end{theorem}
\begin{proof}
反设 $A>B$。取 $\varepsilon=(A-B)/3$。当 $x$ 充分接近 $x_0$ 时，
\[
 f(x)>A-\varepsilon>B+\varepsilon>g(x),
\]
与 $f(x)\le g(x)$ 矛盾。因此 $A\le B$。
\end{proof}

\begin{theorem}[夹逼定理]
若在某个去心邻域内有
\[
 f(x)\le h(x)\le g(x),
\]
且
\[
 \lim_{x\to x_0}f(x)=\lim_{x\to x_0}g(x)=A,
\]
则 $\lim_{x\to x_0}h(x)=A$。
\end{theorem}
\begin{proof}
任取 $\varepsilon>0$。取 $\delta>0$，使 $0<|x-x_0|<\delta$ 时既有夹逼关系，又有
\[
 |f(x)-A|<\varepsilon,\qquad |g(x)-A|<\varepsilon.
\]
于是
\[
 A-\varepsilon<f(x)\le h(x)\le g(x)<A+\varepsilon,
\]
即 $|h(x)-A|<\varepsilon$，故 $h(x)\to A$。
\end{proof}

\begin{example}
证明
\[
 \lim_{x\to0}x\left\lfloor\frac1x\right\rfloor=1.
\]
\end{example}
\begin{proof}
先设 $x>0$。由取整函数的性质，
\[
 \frac1x-1<\left\lfloor\frac1x\right\rfloor\le\frac1x.
\]
乘以正数 $x$，得
\[
 1-x<x\left\lfloor\frac1x\right\rfloor\le1.
\]
当 $x\to0^+$ 时，由夹逼定理，右侧函数趋于 $1$。当 $x<0$ 时，同样由
\[
 \frac1x-1<\left\lfloor\frac1x\right\rfloor\le\frac1x
\]
乘以负数 $x$ 得
\[
 1\le x\left\lfloor\frac1x\right\rfloor<1-x,
\]
故右侧也趋于 $1$。再由双侧极限判定，原极限为 $1$。
\end{proof}

\subsection*{四则运算法则}
\begin{theorem}[四则运算法则]
若 $\lim_{x\to x_0}f(x)=A$、$\lim_{x\to x_0}g(x)=B$，则
\[
 \lim_{x\to x_0}[f(x)\pm g(x)]=A\pm B,
\]
\[
 \lim_{x\to x_0}f(x)g(x)=AB.
\]
若 $B\ne0$，则在某个去心邻域内 $g(x)\ne0$，且
\[
 \lim_{x\to x_0}\frac{f(x)}{g(x)}=\frac AB.
\]
\end{theorem}
\begin{proof}
加减法由
\[
 |[f(x)\pm g(x)]-(A\pm B)|
 \le |f(x)-A|+|g(x)-B|
\]
及 $\varepsilon/2$ 的选取立即得到。

乘法中，先由局部有界性取 $M>0$，使 $|f(x)|<M$。于是
\[
 |f(x)g(x)-AB|
 \le |f(x)|\,|g(x)-B|+|B|\,|f(x)-A|
 <M|g(x)-B|+|B|\,|f(x)-A|.
\]
右端两项可同时取得任意小，故乘积极限为 $AB$。

若 $B\ne0$，由 $g(x)\to B$，取 $\varepsilon=|B|/2$，可得
\[
 |g(x)|>\frac{|B|}{2}>0
\]
在某个去心邻域内成立。因此
\[
 \left|\frac1{g(x)}-\frac1B\right|
 =\frac{|g(x)-B|}{|B|\,|g(x)|}
 \le\frac2{|B|^2}|g(x)-B|\to0.
\]
再用乘法法则即得商的结论。
\end{proof}

\begin{example}
计算
\[
 \lim_{x\to+\infty}\frac{2x^2+3x}{x^2+2x+1}.
\]
\end{example}
\begin{proof}[解]
除以 $x^2$ 后为
\[
 \frac{2+3/x}{1+2/x+1/x^2}\longrightarrow2.
\]
\end{proof}

\begin{example}
计算 $\displaystyle\lim_{x\to\pi/4}(x\tan x-1)$。
\end{example}
\begin{proof}[解]
由上一节已经证明的正弦、余弦极限和四则运算法则，
\[
 \lim_{x\to\pi/4}(x\tan x-1)
 =\frac\pi4\cdot\frac{\sin(\pi/4)}{\cos(\pi/4)}-1
 =\frac\pi4-1.
\]
\end{proof}

\subsection*{归结原则}
直接从函数极限定义判断，有时需要同时处理所有充分接近 $x_0$ 的点。归结原则把它转化为数列问题。

\begin{theorem}[归结原则（Heine 定理）]
设 $f$ 在 $x_0$ 的某个去心邻域 $\U^0(x_0,\delta_0)$ 内有定义，则
\[
 \lim_{x\to x_0}f(x)=A
\]
当且仅当对任意满足
\[
 x_n\in\U^0(x_0,\delta_0),\qquad x_n\to x_0
\]
的数列 $\{x_n\}$，都有
\[
 f(x_n)\to A.
\]
\end{theorem}
\begin{proof}
先证必要性。设 $f(x)\to A$，任取 $x_n\in\U^0(x_0,\delta_0)$ 且 $x_n\to x_0$。给定 $\varepsilon>0$，由函数极限定义取 $\delta>0$，使
\[
 0<|x-x_0|<\delta\Longrightarrow |f(x)-A|<\varepsilon.
\]
又由 $x_n\to x_0$，存在 $N$，使 $n>N$ 时 $|x_n-x_0|<\delta$。因 $x_n\ne x_0$，故 $|f(x_n)-A|<\varepsilon$，即 $f(x_n)\to A$。

再证充分性。反设函数极限不等于 $A$。则存在 $\varepsilon_0>0$，使任意 $\delta>0$ 都能找到满足 $0<|x-x_0|<\delta$ 的点 $x$，使
\[
 |f(x)-A|\ge\varepsilon_0.
\]
对每个 $n\in\Nplus$ 取 $x_n$ 满足
\[
 0<|x_n-x_0|<\min\left\{\delta_0,\frac1n\right\},\qquad |f(x_n)-A|\ge\varepsilon_0.
\]
则 $x_n\in\U^0(x_0,\delta_0)$ 且 $x_n\to x_0$，但 $f(x_n)$ 不趋于 $A$，与假设矛盾。因此 $\lim_{x\to x_0}f(x)=A$。
\end{proof}

\begin{example}
证明 $\displaystyle\lim_{x\to0}\sin(1/x)$ 不存在。
\end{example}
\begin{proof}
取
\[
 x_n=\frac1{2\pi n+\pi/2},\qquad
 y_n=\frac1{2\pi n+3\pi/2}.
\]
两列都趋于 $0$，但
\[
 \sin\frac1{x_n}=1,\qquad \sin\frac1{y_n}=-1.
\]
若 $\lim_{x\to0}\sin(1/x)$ 存在，则由归结原则，两列函数值应趋于同一个极限，矛盾。故该极限不存在。
\end{proof}

归结原则中的“任意数列”不能改成“某个数列”。一列函数值收敛，只能确定沿这一列点趋近时的函数值极限；要证明函数极限存在，必须保证所有允许的趋近数列都给出同一个极限。反过来，只要找到一列函数值不收敛，或者两列函数值趋于不同的数，就能否定有限极限的存在。

\subsection*{单侧归结原则}
归结原则的证明也适用于单侧趋近和无穷远处。对于单侧极限，还可以把需要检验的数列限制为单调趋近的数列。

\begin{theorem}[右极限的归结原则]
设 $f$ 在 $(x_0,x_0+\delta_0)$ 上有定义，则
\[
 \lim_{x\to x_0^+}f(x)=A
\]
当且仅当对任意满足
\[
 x_0<x_n<x_0+\delta_0,\qquad x_n\downarrow x_0
\]
的严格递减数列，都有 $f(x_n)\to A$。
\end{theorem}
\begin{proof}
必要性由归结原则直接得到。反之，若右极限不等于 $A$，则存在 $\varepsilon_0>0$，使每个充分小的右邻域中都有点 $x$ 满足 $|f(x)-A|\ge\varepsilon_0$。

先取这样的点 $x_1\in(x_0,x_0+\delta_0)$。已取到 $x_n$ 后，再取一个这样的点 $x_{n+1}$，使
\[
 0<x_{n+1}-x_0<\min\left\{\frac1{n+1},\frac{x_n-x_0}{2}\right\}.
\]
这样得到严格递减且趋于 $x_0$ 的数列，但始终有 $|f(x_n)-A|\ge\varepsilon_0$，与假设矛盾。
\end{proof}

左极限可相应地只检验严格递增到 $x_0$ 的数列；对于 $x\to+\infty$，可只检验严格递增且趋于 $+\infty$ 的数列。这里限制的是自变量数列的单调性，并不要求函数 $f$ 单调。

\subsection*{单调有界定理}
归结原则把函数极限转化为数列极限，但还需要判断各列函数值是否趋向同一个数。对单调函数，可以直接借助确界原理确定单侧极限。

\begin{theorem}[单调有界定理]
设 $f$ 在 $x_0$ 的某个右邻域内单调且有界，则有限右极限 $\lim_{x\to x_0^+}f(x)$ 存在。具体地，在这个右邻域内，
\[
 \begin{array}{c|c}
 f\text{ 递增}&\displaystyle\lim_{x\to x_0^+}f(x)=\inf f\\[0.5em]
 f\text{ 递减}&\displaystyle\lim_{x\to x_0^+}f(x)=\sup f
 \end{array}
\]
其中 $\inf f$、$\sup f$ 分别指该邻域上函数值集合的下确界、上确界。对于左极限，递增函数对应上确界，递减函数对应下确界。
\end{theorem}
\begin{proof}
以 $f$ 在 $(x_0,x_0+\delta_0)$ 上递增为例。令
\[
 A=\inf\{f(x):x_0<x<x_0+\delta_0\}.
\]
任取 $\varepsilon>0$，由下确界的定义，存在 $x'\in(x_0,x_0+\delta_0)$，使 $f(x')<A+\varepsilon$。取 $\delta=x'-x_0$。当 $x_0<x<x_0+\delta=x'$ 时，由单调性，
\[
 A\le f(x)\le f(x')<A+\varepsilon,
\]
故 $|f(x)-A|<\varepsilon$。这就证明了右极限为 $A$。其余情形同理。
\end{proof}

判定某一侧的极限时，只需考察这一侧某个足够小邻域内的单调性和有界性；左右两侧的极限仍可能不同。反过来，有限单侧极限存在必然推出这一侧的局部有界性。因此，对已经单调的函数，有限单侧极限存在与在某个更小的单侧邻域内有界等价，不要求它在原来的整个区间上有界。

\subsection*{函数极限的柯西准则}
还有一种判别方法只比较邻近的函数值，不预先指定极限值。

\begin{theorem}[柯西准则]
设 $f$ 在 $\U^0(x_0,\delta_0)$ 上有定义，则 $\lim_{x\to x_0}f(x)$ 存在有限值的充要条件是：对任意 $\varepsilon>0$，存在 $0<\delta\le\delta_0$，使任意两点 $x',x''\in\U^0(x_0,\delta)$ 都满足
\[
 |f(x')-f(x'')|<\varepsilon.
\]
\end{theorem}
\begin{proof}
必要性：设 $f(x)\to A$。任取 $\varepsilon>0$，由极限定义取 $\delta>0$，使去心邻域内的 $x$ 满足 $|f(x)-A|<\varepsilon/2$。于是其中任意两点满足
\[
 |f(x')-f(x'')|
 \le |f(x')-A|+|f(x'')-A|<\varepsilon.
\]

充分性：先固定一列 $x_n\in\U^0(x_0,\delta_0)$，使 $x_n\to x_0$。任取 $\varepsilon>0$，按题设取 $\delta>0$。当 $n,m$ 充分大时，$x_n,x_m$ 都在 $\U^0(x_0,\delta)$ 中，因而
\[
 |f(x_n)-f(x_m)|<\varepsilon.
\]
由数列的柯西收敛准则，存在 $A\in\R$，使 $f(x_n)\to A$。

下面用同一个 $A$ 控制任意邻近点的函数值。给定 $\varepsilon>0$，按题设取 $\delta>0$，使该去心邻域内任意两点的函数值之差小于 $\varepsilon/2$。再固定一个充分大的下标 $n_0$，使
\[
 x_{n_0}\in\U^0(x_0,\delta),\qquad
 |f(x_{n_0})-A|<\frac\varepsilon2.
\]
这样，对任意 $x\in\U^0(x_0,\delta)$，都有
\[
 |f(x)-A|\le |f(x)-f(x_{n_0})|+|f(x_{n_0})-A|
 <\frac\varepsilon2+\frac\varepsilon2=\varepsilon.
\]
故 $\lim_{x\to x_0}f(x)=A$。
\end{proof}

对于 $x\to+\infty$，只需把两个点都位于去心邻域的条件换成 $x',x''>M$。相应地，有限极限不存在当且仅当存在 $\varepsilon_0>0$，使任意远处都能找到两点，其函数值之差不小于 $\varepsilon_0$。有限点处的否定形式则为
\[
 \exists\varepsilon_0>0,\ \forall\delta\in(0,\delta_0],\
 \exists x',x''\in\U^0(x_0,\delta):\quad
 |f(x')-f(x'')|\ge\varepsilon_0.
\]

\begin{example}
用归结原则和柯西准则分别证明 $\displaystyle\lim_{x\to+\infty}\sin x$ 不存在。
\end{example}
\begin{proof}
取 $x_n=2n\pi$、$y_n=2n\pi+\pi/2$。两列都趋于 $+\infty$，而
\[
 \sin x_n=0,\qquad \sin y_n=1.
\]
由归结原则，原极限不存在。若用柯西准则，则取 $\varepsilon_0=1$：对任意 $M>0$，取充分大的 $n$，便有 $x_n,y_n>M$，而 $|\sin x_n-\sin y_n|=1$，不满足柯西条件。
\end{proof}

\begin{example}
证明函数 $f(x)=1/x$ 在 $x\to0$ 时不存在有限极限。
\end{example}
\begin{proof}
取 $x_n=1/n$、$y_n=1/(n+1)$。两列都趋于 $0$，且
\[
 \left|\frac1{x_n}-\frac1{y_n}\right|=|n-(n+1)|=1.
\]
因此在任意小的去心邻域内，均有两个函数值之差等于 $1$，由柯西准则，有限极限不存在。

也可只取 $x_n=1/n$：对应的函数值 $n$ 不收敛于有限实数，由归结原则即得结论。这里否定的是有限极限；无穷大量的含义将在后面另行定义。
\end{proof}
